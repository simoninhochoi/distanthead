% ---- structure ------------------------------------------------------------
% A section head: small-caps, a rule under it, air above.
\newcommand{\cvsection}[1]{%
  \vspace{1.0em}%
  {\bfseries\scshape #1}\par
  \vspace{0.25em}\hrule height 0.5pt\vspace{0.7em}}

% A subsection inside a section (Peer-Reviewed Articles, Methodology, …).
\newcommand{\cvsub}[1]{\vspace{0.5em}{\itshape #1}\par\vspace{0.35em}}

% One line with the date pushed to the right margin.
\newcommand{\dated}[2]{#1\hfill #2\par}

% An employment or education block: organisation, then indented roles.
\newenvironment{roles}{\vspace{0.15em}\begin{list}{}{%
  \setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0pt}%
  \setlength{\parsep}{0pt}\setlength{\topsep}{0pt}}\item[]}{\end{list}}

% Push a year to the right margin, letting it drop to its own right-aligned
% line when the text runs to the edge. A bare \hfill has no room to expand
% there and the year ends up welded to the last word (Hawaii Press.forthcoming).
\newcommand{\rightdate}[1]{%
  \unskip\nobreak\hfil\penalty50\hskip1.5em\null\nobreak\hfil #1%
  \parfillskip=0pt \finalhyphendemerits=0 \par}

% A flowing entry (a publication, a talk) with a hanging indent and a year.
\newenvironment{hang}{%
  \begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemindent}{-1.5em}%
    \setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\topsep}{0pt}}%
  \item[]}{\end{list}\vspace{0.35em}}
\newcommand{\bibentry}[2]{\begin{hang}#1\rightdate{#2}\end{hang}}
\newcommand{\bibplain}[1]{\begin{hang}#1\end{hang}}
